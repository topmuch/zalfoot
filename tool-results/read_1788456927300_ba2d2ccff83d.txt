    62→function todayIso(): string {
    63→  const d = new Date()
    64→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    65→}

    67→export function BookingDialog({
    68→  open,
    69→  onOpenChange,
    70→  facilities,
    71→  preselectedFacilityId,
    72→}: {
    73→  open: boolean
    74→  onOpenChange: (open: boolean) => void
    75→  facilities: Facility[]
    76→  preselectedFacilityId?: string
    77→}) {
    78→  const { toast } = useToast()
    79→  const [submitting, setSubmitting] = useState(false)
    80→  const [form, setForm] = useState<BookingForm>({
    81→    facilityId: preselectedFacilityId ?? '',
    82→    date: todayIso(),
    83→    startTime: '18:00',
    84→    endTime: '19:00',
    85→    customerName: '',
    86→    customerEmail: '',
    87→    customerPhone: '',
    88→    notes: '',
    89→  })
    90→
    91→  const selected = facilities.find((f) => f.id === form.facilityId)
    92→  const set = (patch: Partial<BookingForm>) => setForm((prev) => ({ ...prev, ...patch }))
    93→
    94→  // À chaque ouverture, applique le terrain préselectionné (ou réinitialise)
    95→  useEffect(() => {
    96→    if (open) {
    97→      setForm((prev) => ({ ...prev, facilityId: preselectedFacilityId ?? '' }))
    98→    }
    99→  }, [open, preselectedFacilityId])
   100→
   101→  const canSubmit =
   102→    form.facilityId &&
   103→    form.date &&
   104→    form.startTime &&
   105→    form.endTime &&
   106→    form.customerName.trim().length > 1 &&
   107→    form.endTime > form.startTime
   108→
   109→  async function handleSubmit(e: React.FormEvent) {
   110→    e.preventDefault()
   111→    if (!canSubmit || submitting) return
   112→    setSubmitting(true)
   113→    try {
   114→      const result = await apiFetch<{ reservation: { reference: string } }>('/api/reservations', {
   115→        method: 'POST',
   116→        body: {
   117→          facilityId: form.facilityId,
   118→          date: form.date,
   119→          startTime: form.startTime,
   120→          endTime: form.endTime,
   121→          customerName: form.customerName.trim(),
   122→          customerEmail: form.customerEmail.trim() || null,
   123→          customerPhone: form.customerPhone.trim() || null,
   124→          notes: form.notes.trim() || null,
   125→          source: 'PUBLIC',
   126→        },
   127→      })
   128→      toast({
   129→        title: 'Réservation envoyée ! 🎉',
   130→        description: `Référence ${result.reservation.reference}. Vous recevrez une confirmation sous 24 h.`,
   131→      })
   132→      onOpenChange(false)
   133→      setForm((prev) => ({ ...prev, customerName: '', customerEmail: '', customerPhone: '', notes: '' }))
   134→    } catch (error) {
   135→      const message = error instanceof ApiError ? error.message : 'Une erreur est survenue. Réessayez.'
   136→      toast({ title: 'Échec de la réservation', description: message, variant: 'destructive' })
   137→    } finally {
   138→      setSubmitting(false)
   139→    }
   140→  }
   141→
   142→  return (
   143→    <Dialog open={open} onOpenChange={onOpenChange}>
   144→      <DialogContent className="sm:max-w-lg zalspor-scroll max-h-[90vh] overflow-y-auto">
   145→        <DialogHeader>
   146→          <DialogTitle className="flex items-center gap-2 text-xl">
   147→            <CalendarCheck className="size-5 text-primary" />
   148→            Réserver un terrain
   149→          </DialogTitle>
   150→          <DialogDescription>
   151→            Choisissez votre terrain de football, votre créneau horaire et laissez vos coordonnées.
   152→            Confirmation sous 24 h.
   153→          </DialogDescription>
   154→        </DialogHeader>
   155→
   156→        <form onSubmit={handleSubmit} className="grid gap-4">
   157→          <div className="grid gap-2">
   158→            <Label htmlFor="booking-facility">Terrain *</Label>
   159→            <Select value={form.facilityId} onValueChange={(v) => set({ facilityId: v })}>
   160→              <SelectTrigger id="booking-facility">
   161→                <SelectValue placeholder="Sélectionnez un terrain" />
   162→              </SelectTrigger>
   163→              <SelectContent>
   164→                {facilities.map((f) => (
   165→                  <SelectItem key={f.id} value={f.id}>
   166→                    {f.name} — {formatPrice(f.pricePerHour)}/h
   167→                  </SelectItem>
   168→                ))}
   169→              </SelectContent>
   170→            </Select>
   171→          </div>
   172→
   173→          <div className="grid grid-cols-3 gap-3">
   174→            <div className="grid gap-2">
   175→              <Label htmlFor="booking-date">Date *</Label>
   176→              <Input
   177→                id="booking-date"
   178→                type="date"
   179→                min={todayIso()}
   180→                value={form.date}
   181→                onChange={(e) => set({ date: e.target.value })}
   182→                required
   183→              />
   184→            </div>
   185→            <div className="grid gap-2">
   186→              <Label htmlFor="booking-start">Début *</Label>
   187→              <Input
   188→                id="booking-start"
   189→                type="time"
   190→                value={form.startTime}
   191→                onChange={(e) => set({ startTime: e.target.value })}
   192→                required
   193→              />
   194→            </div>
   195→            <div className="grid gap-2">
   196→              <Label htmlFor="booking-end">Fin *</Label>
   197→              <Input
   198→                id="booking-end"
   199→                type="time"
   200→                value={form.endTime}
   201→                onChange={(e) => set({ endTime: e.target.value })}
   202→                required
   203→              />
   204→            </div>
   205→          </div>
   206→
   207→          <div className="grid gap-2">
   208→            <Label htmlFor="booking-name">Nom complet *</Label>
   209→            <Input
   210→              id="booking-name"
   211→              placeholder="Ex. Aïssatou Diallo"
   212→              value={form.customerName}
   213→              onChange={(e) => set({ customerName: e.target.value })}
   214→              required
   215→              minLength={2}
   216→            />
   217→          </div>
   218→
   219→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   220→            <div className="grid gap-2">
   221→              <Label htmlFor="booking-email">E-mail</Label>
   222→              <Input
   223→                id="booking-email"
   224→                type="email"
   225→                placeholder="vous@exemple.com"
   226→                value={form.customerEmail}
   227→                onChange={(e) => set({ customerEmail: e.target.value })}
   228→              />
   229→            </div>
   230→            <div className="grid gap-2">
   231→              <Label htmlFor="booking-phone">Téléphone</Label>
   232→              <Input
   233→                id="booking-phone"
   234→                placeholder="+221 77 000 00 00"
   235→                value={form.customerPhone}
   236→                onChange={(e) => set({ customerPhone: e.target.value })}
   237→              />
   238→            </div>
   239→          </div>
   240→
   241→          <div className="grid gap-2">
   242→            <Label htmlFor="booking-notes">Notes (optionnel)</Label>
   243→            <Textarea
   244→              id="booking-notes"
   245→              placeholder="Précisions, équipement souhaité…"
   246→              rows={2}
   247→              value={form.notes}
   248→              onChange={(e) => set({ notes: e.target.value })}
   249→            />
   250→          </div>
   251→
   252→          {selected && (
   253→            <div className="rounded-lg bg-muted/60 border px-4 py-3 flex items-center justify-between text-sm">
   254→              <span className="text-muted-foreground flex items-center gap-2">
   255→                <Clock className="size-4" />
   256→                Tarif : <strong className="text-foreground">{formatPrice(selected.pricePerHour)}/heure</strong>
   257→              </span>
   258→              <span className="text-muted-foreground flex items-center gap-1.5">
   259→                <Users className="size-4" />
   260→                {selected.capacity} joueurs
   261→              </span>
   262→            </div>
   263→          )}
   264→
   265→          <DialogFooter className="mt-1">
   266→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   267→              Annuler
   268→            </Button>
   269→            <Button type="submit" disabled={!canSubmit || submitting}>
   270→              {submitting ? <Loader2 className="size-4 animate-spin" /> : <CheckCircle2 className="size-4" />}
   271→              {submitting ? 'Envoi…' : 'Confirmer la réservation'}
   272→            </Button>
   273→          </DialogFooter>
   274→        </form>
   275→      </DialogContent>
   276→    </Dialog>
   277→  )
   278→}
