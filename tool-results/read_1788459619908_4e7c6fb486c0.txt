     1→'use client'
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
     4→import {
     5→  ArrowLeft,
     6→  CalendarCheck,
     7→  CheckCircle2,
     8→  ChevronLeft,
     9→  ChevronRight,
    10→  Clock,
    11→  Info,
    12→  Loader2,
    13→  Phone,
    14→  User,
    15→  Users,
    16→  Waves,
    17→} from 'lucide-react'
    18→import {
    19→  addMonths,
    20→  eachDayOfInterval,
    21→  endOfMonth,
    22→  endOfWeek,
    23→  format,
    24→  isSameDay,
    25→  isSameMonth,
    26→  startOfMonth,
    27→  startOfWeek,
    28→} from 'date-fns'
    29→import { fr } from 'date-fns/locale'
    30→import { Button } from '@/components/ui/button'
    31→import { Card, CardContent } from '@/components/ui/card'
    32→import { Input } from '@/components/ui/input'
    33→import { Label } from '@/components/ui/label'
    34→import { Skeleton } from '@/components/ui/skeleton'
    35→import {
    36→  Dialog,
    37→  DialogContent,
    38→  DialogDescription,
    39→  DialogFooter,
    40→  DialogHeader,
    41→  DialogTitle,
    42→} from '@/components/ui/dialog'
    43→import { useToast } from '@/hooks/use-toast'
    44→import { apiFetch, ApiError } from './api'
    45→import {
    46→  buildWaveUrl,
    47→  formatDateFr,
    48→  formatHourLabel,
    49→  formatPrice,
    50→  type AvailabilitySlot,
    51→  type DayAvailability,
    52→  type Facility,
    53→  type MonthAvailability,
    54→  type PublicSettings,
    55→  type Reservation,
    56→} from './types'
    57→
    58→// ============================================================
    59→// Réservation client : calendrier visible + créneaux horaires
    60→// 08:00 → 00:00 (minuit), 1 h par créneau.
    61→// Étape 1 : terrain + date + créneaux · Étape 2 : coordonnées
    62→// + paiement Wave · Étape 3 : confirmation avec référence.
    63→// ============================================================
    64→
    65→const WEEKDAYS = ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim']
    66→/** Réservation possible jusqu'à 60 jours à l'avance */
    67→const MAX_DAYS_AHEAD = 60
    68→
    69→function isoOf(d: Date): string {
    70→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    71→}
    72→
    73→function monthKey(d: Date): string {
    74→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
    75→}
    76→
    77→type Selection = { start: number; end: number }
    78→
    79→type Step = 'slot' | 'details' | 'done'
    80→
    81→export function BookingDialog({
    82→  open,
    83→  onOpenChange,
    84→  facilities,
    85→  preselectedFacilityId,
    86→}: {
    87→  open: boolean
    88→  onOpenChange: (open: boolean) => void
    89→  facilities: Facility[]
    90→  preselectedFacilityId?: string
    91→}) {
    92→  const { toast } = useToast()
    93→
    94→  const [step, setStep] = useState<Step>('slot')
    95→  const [facilityId, setFacilityId] = useState('')
    96→  const [date, setDate] = useState(() => isoOf(new Date()))
    97→  const [monthCursor, setMonthCursor] = useState(() => startOfMonth(new Date()))
    98→  const [slots, setSlots] = useState<AvailabilitySlot[]>([])
    99→  const [slotsLoading, setSlotsLoading] = useState(false)
   100→  const [monthSummary, setMonthSummary] = useState<Record<string, number> | null>(null)
   101→  const [selection, setSelection] = useState<Selection | null>(null)
   102→  const [customerName, setCustomerName] = useState('')
   103→  const [customerPhone, setCustomerPhone] = useState('')
   104→  const [submitting, setSubmitting] = useState(false)
   105→  const [created, setCreated] = useState<Reservation | null>(null)
   106→  const [waveLink, setWaveLink] = useState<string | null>(null)
   107→
   108→  const fetchSeq = useRef(0)
   109→
   110→  const facility = useMemo(
   111→    () => facilities.find((f) => f.id === facilityId),
   112→    [facilities, facilityId],
   113→  )
   114→  const todayIso = isoOf(new Date())
   115→  const maxIso = useMemo(() => {
   116→    const d = new Date()
   117→    d.setDate(d.getDate() + MAX_DAYS_AHEAD)
   118→    return isoOf(d)
   119→  }, [])
   120→
   121→  // ===== Ouverture : terrain présélectionné + réinitialisation =====
   122→  useEffect(() => {
   123→    if (open) {
   124→      setStep('slot')
   125→      setSelection(null)
   126→      setCreated(null)
   127→      setDate(isoOf(new Date()))
   128→      setMonthCursor(startOfMonth(new Date()))
   129→      setFacilityId(preselectedFacilityId ?? facilities[0]?.id ?? '')
   130→      // Lien de paiement Wave (configuré par l'admin)
   131→      apiFetch<PublicSettings>('/api/settings')
   132→        .then((s) => setWaveLink(s.wavePaymentLink))
   133→        .catch(() => setWaveLink(null))
   134→    }
   135→  }, [open, preselectedFacilityId, facilities])
   136→
   137→  // ===== Créneaux du jour sélectionné =====
   138→  const loadSlots = useCallback(async (fid: string, day: string) => {
   139→    if (!fid || !day) return
   140→    const seq = ++fetchSeq.current
   141→    setSlotsLoading(true)
   142→    try {
   143→      const data = await apiFetch<DayAvailability>(
   144→        `/api/availability?facilityId=${encodeURIComponent(fid)}&date=${day}`,
   145→      )
   146→      if (seq === fetchSeq.current) setSlots(data.slots)
   147→    } catch {
   148→      if (seq === fetchSeq.current) setSlots([])
   149→    } finally {
   150→      if (seq === fetchSeq.current) setSlotsLoading(false)
   151→    }
   152→  }, [])
   153→
   154→  useEffect(() => {
   155→    if (open) {
   156→      setSelection(null)
   157→      loadSlots(facilityId, date)
   158→    }
   159→  }, [open, facilityId, date, loadSlots])
   160→
   161→  // ===== Résumé mensuel (calendrier : jours complets / bientôt complets) =====
   162→  useEffect(() => {
   163→    if (!open || !facilityId) return
   164→    let cancelled = false
   165→    apiFetch<MonthAvailability>(
   166→      `/api/availability?facilityId=${encodeURIComponent(facilityId)}&month=${monthKey(monthCursor)}`,
   167→    )
   168→      .then((data) => {
   169→        if (!cancelled) setMonthSummary(data.days)
   170→      })
   171→      .catch(() => {
   172→        if (!cancelled) setMonthSummary(null)
   173→      })
   174→    return () => {
   175→      cancelled = true
   176→    }
   177→  }, [open, facilityId, monthCursor])
   178→
   179→  // ===== Sélection de créneaux consécutifs =====
   180→  function toggleSlot(idx: number) {
   181→    const slot = slots[idx]
   182→    if (!slot || slot.state !== 'FREE') return
   183→    if (!selection) {
   184→      setSelection({ start: idx, end: idx })
   185→      return
   186→    }
   187→    if (idx >= selection.start && idx <= selection.end) {
   188→      if (idx === selection.start && idx === selection.end) setSelection(null)
   189→      else if (idx === selection.start) setSelection({ start: idx + 1, end: selection.end })
   190→      else if (idx === selection.end) setSelection({ start: selection.start, end: idx - 1 })
   191→      return
   192→    }
   193→    if (idx === selection.start - 1) setSelection({ start: idx, end: selection.end })
   194→    else if (idx === selection.end + 1) setSelection({ start: selection.start, end: idx })
   195→    else setSelection({ start: idx, end: idx })
   196→  }
   197→
   198→  const selectedSlots = useMemo(
   199→    () => (selection ? slots.slice(selection.start, selection.end + 1) : []),
   200→    [selection, slots],
   201→  )
   202→  const hours = selection ? selection.end - selection.start + 1 : 0
   203→  const total = facility && hours > 0 ? hours * facility.pricePerHour : 0
   204→  const canContinue = hours > 0 && !!facility
   205→
   206→  // ===== Jours du calendrier =====
   207→  const calendarDays = useMemo(() => {
   208→    const start = startOfWeek(startOfMonth(monthCursor), { weekStartsOn: 1 })
   209→    const end = endOfWeek(endOfMonth(monthCursor), { weekStartsOn: 1 })
   210→    return eachDayOfInterval({ start, end })
   211→  }, [monthCursor])
   212→
   213→  const prevMonthDisabled = isSameMonth(monthCursor, new Date())
   214→  const nextMonthDisabled = isSameMonth(monthCursor, addMonths(new Date(), 2))
   215→
   216→  // ===== Soumission =====
   217→  const canSubmit =
   218→    step === 'details' &&
   219→    !!facility &&
   220→    hours > 0 &&
   221→    customerName.trim().length > 1 &&
   222→    customerPhone.replace(/\D/g, '').length >= 8
   223→
   224→  async function handleSubmit(e: React.FormEvent) {
   225→    e.preventDefault()
   226→    if (!canSubmit || submitting || !selection || !facility) return
   227→    const startSlot = slots[selection.start]
   228→    const endSlot = slots[selection.end]
   229→    if (!startSlot || !endSlot) return
   230→    setSubmitting(true)
   231→    try {
   232→      const result = await apiFetch<{ reservation: Reservation }>('/api/reservations', {
   233→        method: 'POST',
   234→        body: {
   235→          facilityId: facility.id,
   236→          date,
   237→          startTime: startSlot.start,
   238→          endTime: endSlot.end,
   239→          customerName: customerName.trim(),
   240→          customerPhone: customerPhone.trim(),
   241→          source: 'PUBLIC',
   242→        },
   243→      })
   244→      setCreated(result.reservation)
   245→      setStep('done')
   246→      toast({
   247→        title: 'Réservation enregistrée ! 🎉',
   248→        description: `Référence ${result.reservation.reference.slice(0, 10).toUpperCase()}. Effectuez le paiement Wave pour la confirmer.`,
   249→      })
   250→    } catch (error) {
   251→      const message = error instanceof ApiError ? error.message : 'Une erreur est survenue. Réessayez.'
   252→      toast({ title: 'Échec de la réservation', description: message, variant: 'destructive' })
   253→      // Conflit ou créneau devenu indisponible → retour au choix du créneau
   254→      if (error instanceof ApiError && (error.status === 409 || error.status === 400)) {
   255→        setStep('slot')
   256→        setSelection(null)
   257→        loadSlots(facilityId, date)
   258→      }
   259→    } finally {
   260→      setSubmitting(false)
   261→    }
   262→  }
   263→
   264→  function payWithWave() {
   265→    if (!waveLink || !created) return
   266→    const url = buildWaveUrl(waveLink, { amount: created.amount ?? 0, reference: created.reference })
   267→    window.open(url, '_blank', 'noopener')
   268→  }
   269→
   270→  // ===== Rendu =====
   271→  const stepTitles: Record<Step, { title: string; desc: string }> = {
   272→    slot: {
   273→      title: 'Réserver votre créneau',
   274→      desc: 'Choisissez le terrain, la date puis votre créneau horaire (08:00 → minuit).',
   275→    },
   276→    details: {
   277→      title: 'Vos coordonnées',
   278→      desc: 'Laissez votre nom et votre numéro de téléphone, puis payez avec Wave.',
   279→    },
   280→    done: { title: 'Réservation enregistrée', desc: 'Votre créneau est bloqué, il ne reste qu’à payer.' },
   281→  }
   282→
   283→  const freeCount = slots.filter((s) => s.state === 'FREE').length
   284→  const noSlotsLeft = !slotsLoading && slots.length > 0 && freeCount === 0
   285→
   286→  return (
   287→    <Dialog open={open} onOpenChange={onOpenChange}>
   288→      <DialogContent className="sm:max-w-lg zalspor-scroll max-h-[92vh] overflow-y-auto">
   289→        <DialogHeader>
   290→          <DialogTitle className="flex items-center gap-2 text-xl">
   291→            <CalendarCheck className="size-5 text-primary" />
   292→            {stepTitles[step].title}
   293→          </DialogTitle>
   294→          <DialogDescription>{stepTitles[step].desc}</DialogDescription>
   295→        </DialogHeader>
   296→
   297→        {/* ============ ÉTAPE 1 : terrain + calendrier + créneaux ============ */}
   298→        {step === 'slot' && (
   299→          <div className="grid gap-5">
   300→            {facilities.length > 1 ? (
   301→              <div className="grid grid-cols-1 gap-2">
   302→                <Label>Terrain</Label>
   303→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-2">
   304→                  {facilities.map((f) => (
   305→                    <button
   306→                      key={f.id}
   307→                      type="button"
   308→                      onClick={() => setFacilityId(f.id)}
   309→                      className={`rounded-xl border px-3 py-2.5 text-left transition-all cursor-pointer ${
   310→                        facilityId === f.id
   311→                          ? 'border-primary bg-primary/5 shadow-sm'
   312→                          : 'hover:border-primary/40'
   313→                      }`}
   314→                      aria-pressed={facilityId === f.id}
   315→                    >
   316→                      <p className="text-sm font-semibold leading-tight truncate">{f.name.split('—')[0].trim()}</p>
   317→                      <p className="text-xs text-muted-foreground mt-1">
   318→                        {formatPrice(f.pricePerHour)}/h · {f.capacity} joueurs
   319→                      </p>
   320→                    </button>
   321→                  ))}
   322→                </div>
   323→              </div>
   324→            ) : facilities.length === 1 ? (
   325→              <Card>
   326→                <CardContent className="flex items-center justify-between py-3 px-4">
   327→                  <span className="text-sm font-semibold">{facilities[0].name}</span>
   328→                  <span className="text-sm text-primary font-bold">{formatPrice(facilities[0].pricePerHour)}/h</span>
   329→                </CardContent>
   330→              </Card>
   331→            ) : (
   332→              <p className="text-sm text-muted-foreground">Aucun terrain disponible pour le moment.</p>
   333→            )}
   334→
   335→            {/* ----- Calendrier mensuel ----- */}
   336→            <div className="rounded-xl border bg-card">
   337→              <div className="flex items-center justify-between px-3 py-2.5 border-b">
   338→                <Button
   339→                  type="button"
   340→                  variant="ghost"
   341→                  size="icon"
   342→                  className="size-8"
   343→                  disabled={prevMonthDisabled}
   344→                  onClick={() => setMonthCursor((m) => addMonths(m, -1))}
   345→                  aria-label="Mois précédent"
   346→                >
   347→                  <ChevronLeft className="size-4" />
   348→                </Button>
   349→                <p className="text-sm font-semibold capitalize">
   350→                  {format(monthCursor, 'MMMM yyyy', { locale: fr })}
   351→                </p>
   352→                <Button
   353→                  type="button"
   354→                  variant="ghost"
   355→                  size="icon"
   356→                  className="size-8"
   357→                  disabled={nextMonthDisabled}
   358→                  onClick={() => setMonthCursor((m) => addMonths(m, 1))}
   359→                  aria-label="Mois suivant"
   360→                >
   361→                  <ChevronRight className="size-4" />
   362→                </Button>
   363→              </div>
   364→              <div className="p-3">
   365→                <div className="grid grid-cols-7 gap-1 mb-1">
   366→                  {WEEKDAYS.map((d) => (
   367→                    <p key={d} className="text-center text-[11px] font-medium text-muted-foreground py-1">
   368→                      {d}
   369→                    </p>
   370→                  ))}
   371→                </div>
   372→                <div className="grid grid-cols-7 gap-1">
   373→                  {calendarDays.map((day) => {
   374→                    const iso = isoOf(day)
   375→                    const inMonth = isSameMonth(day, monthCursor)
   376→                    const past = iso < todayIso
   377→                    const tooFar = iso > maxIso
   378→                    const full = monthSummary !== null && (monthSummary[iso] ?? -1) === 0 && !past && !tooFar
   379→                    const almostFull =
   380→                      !full && monthSummary !== null && (monthSummary[iso] ?? 16) > 0 && (monthSummary[iso] ?? 16) <= 4
   381→                    const disabled = !inMonth || past || tooFar || full
   382→                    const selected = date === iso
   383→                    const today = isSameDay(day, new Date())
   384→                    return (
   385→                      <button
   386→                        key={iso}
   387→                        type="button"
   388→                        disabled={disabled}
   389→                        onClick={() => setDate(iso)}
   390→                        className={`relative h-10 rounded-lg text-sm transition-colors ${
   391→                          selected
   392→                            ? 'bg-primary text-primary-foreground font-bold shadow-sm cursor-pointer'
   393→                            : disabled
   394→                              ? 'text-muted-foreground/40 cursor-not-allowed'
   395→                              : 'hover:bg-primary/10 cursor-pointer'
   396→                        } ${full ? 'line-through' : ''} ${today && !selected ? 'ring-1 ring-primary/50 font-semibold' : ''}`}
   397→                        aria-label={`Choisir le ${format(day, 'd MMMM', { locale: fr })}${full ? ' (complet)' : ''}`}
   398→                        aria-pressed={selected}
   399→                      >
   400→                        {format(day, 'd')}
   401→                        {almostFull && !selected ? (
   402→                          <span
   403→                            className="absolute bottom-1 left-1/2 -translate-x-1/2 size-1 rounded-full bg-amber-500"
   404→                            aria-hidden
   405→                          />
   406→                        ) : null}
   407→                      </button>
   408→                    )
   409→                  })}
   410→                </div>
   411→              </div>
   412→            </div>
   413→
   414→            {/* ----- Grille des créneaux 08:00 → minuit ----- */}
   415→            <div className="grid gap-2">
   416→              <div className="flex items-center justify-between gap-2 min-w-0">
   417→                <Label className="flex items-center gap-1.5 min-w-0">
   418→                  <Clock className="size-3.5 text-primary shrink-0" />
   419→                  <span className="truncate">Créneaux du {formatDateFr(date)}</span>
   420→                </Label>
   421→                <span className="text-xs text-muted-foreground shrink-0 whitespace-nowrap">
   422→                  {slotsLoading ? '…' : `${freeCount} libre${freeCount > 1 ? 's' : ''} / ${slots.length}`}
   423→                </span>
   424→              </div>
   425→
   426→              {slotsLoading ? (
   427→                <div className="grid grid-cols-4 gap-2">
   428→                  {Array.from({ length: 16 }).map((_, i) => (
   429→                    <Skeleton key={i} className="h-16 rounded-xl" />
   430→                  ))}
   431→                </div>
   432→              ) : slots.length === 0 ? (
   433→                <div className="rounded-xl border border-dashed py-8 text-center text-sm text-muted-foreground">
   434→                  Disponibilités indisponibles pour cette date.
   435→                </div>
   436→              ) : (
   437→                <>
   438→                  <div className="grid grid-cols-4 gap-2">
   439→                    {slots.map((slot, idx) => {
   440→                      const isSelected =
   441→                        selection !== null && idx >= selection.start && idx <= selection.end
   442→                      const stateLabel =
   443→                        slot.state === 'BOOKED' ? 'Réservé' : slot.state === 'PAST' ? 'Passé' : slot.state === 'CLOSED' ? 'Fermé' : null
   444→                      return (
   445→                        <button
   446→                          key={slot.start}
   447→                          type="button"
   448→                          disabled={slot.state !== 'FREE'}
   449→                          onClick={() => toggleSlot(idx)}
   450→                          className={`h-16 rounded-xl border flex flex-col items-center justify-center gap-0.5 transition-all ${
   451→                            isSelected
   452→                              ? 'bg-primary border-primary text-primary-foreground shadow-md scale-[1.03]'
   453→                              : slot.state === 'FREE'
   454→                                ? 'hover:border-primary hover:bg-primary/5 cursor-pointer'
   455→                                : 'bg-muted/70 border-transparent text-muted-foreground/60 cursor-not-allowed grayscale'
   456→                          }`}
   457→                          aria-pressed={isSelected}
   458→                          aria-label={`Créneau ${slot.start} – ${slot.end}${stateLabel ? ` (${stateLabel})` : ' (libre)'}`}
   459→                        >
   460→                          <span className="text-sm font-bold tabular-nums">{slot.start}</span>
   461→                          <span className={`text-[10px] leading-none ${isSelected ? 'text-primary-foreground/80' : ''}`}>
   462→                            {stateLabel ?? `${slot.end}`}
   463→                          </span>
   464→                        </button>
   465→                      )
   466→                    })}
   467→                  </div>
   468→                  {noSlotsLeft && (
   469→                    <p className="text-xs text-muted-foreground flex items-center gap-1.5">
   470→                      <Info className="size-3.5 shrink-0" />
   471→                      Plus de créneau disponible ce jour — choisissez une autre date.
   472→                    </p>
   473→                  )}
   474→                </>
   475→              )}
   476→            </div>
   477→
   478→            {/* ----- Récapitulatif + continuation ----- */}
   479→            <div className="rounded-xl bg-muted/50 border px-4 py-3 flex items-center justify-between gap-3 min-h-14">
   480→              {selection && facility ? (
   481→                <>
   482→                  <div className="min-w-0">
   483→                    <p className="text-sm font-semibold truncate">
   484→                      {slots[selection.start]?.start} → {formatHourLabel(slots[selection.end]?.end ?? '')}
   485→                      <span className="text-muted-foreground font-normal"> · {hours} h</span>
   486→                    </p>
   487→                    <p className="text-xs text-muted-foreground truncate">{facility.name}</p>
   488→                  </div>
   489→                  <p className="text-lg font-extrabold text-primary whitespace-nowrap">{formatPrice(total)}</p>
   490→                </>
   491→              ) : (
   492→                <p className="text-sm text-muted-foreground">
   493→                  Sélectionnez un créneau libre (plusieurs heures consécutives possibles).
   494→                </p>
   495→              )}
   496→            </div>
   497→          </div>
   498→        )}
   499→
   500→        {/* ============ ÉTAPE 2 : coordonnées + paiement Wave ============ */}
   501→        {step === 'details' && facility && selection && (
   502→          <form onSubmit={handleSubmit} className="grid gap-4">
   503→            {/* Récap */}
   504→            <Card className="border-primary/30 bg-primary/5">
   505→              <CardContent className="py-4 px-5 grid gap-2">
   506→                <div className="flex items-center justify-between text-sm">
   507→                  <span className="text-muted-foreground flex items-center gap-1.5">
   508→                    <CalendarCheck className="size-4" /> {formatDateFr(date)}
   509→                  </span>
   510→                  <span className="font-bold">{formatDateFr(date).split(' ')[1] ?? ''}</span>
   511→                </div>
   512→                <div className="flex items-center justify-between text-sm">
   513→                  <span className="text-muted-foreground flex items-center gap-1.5">
   514→                    <Clock className="size-4" /> Créneau
   515→                  </span>
   516→                  <span className="font-semibold tabular-nums">
   517→                    {slots[selection.start]?.start} → {formatHourLabel(slots[selection.end]?.end ?? '')} ({hours} h)
   518→                  </span>
   519→                </div>
   520→                <div className="flex items-center justify-between text-sm">
   521→                  <span className="text-muted-foreground flex items-center gap-1.5">
   522→                    <Users className="size-4" /> {facility.name}
