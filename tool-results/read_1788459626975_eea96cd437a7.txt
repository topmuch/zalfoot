    69→    69→function isoOf(d: Date): string {
    70→    70→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    71→    71→}
    72→    72→
    73→    73→function monthKey(d: Date): string {
    74→    74→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
    75→    75→}

    81→    81→export function BookingDialog({
    82→    82→  open,
    83→    83→  onOpenChange,
    84→    84→  facilities,
    85→    85→  preselectedFacilityId,
    86→    86→}: {
    87→    87→  open: boolean
    88→    88→  onOpenChange: (open: boolean) => void
    89→    89→  facilities: Facility[]
    90→    90→  preselectedFacilityId?: string
    91→    91→}) {
    92→    92→  const { toast } = useToast()
    93→    93→
    94→    94→  const [step, setStep] = useState<Step>('slot')
    95→    95→  const [facilityId, setFacilityId] = useState('')
    96→    96→  const [date, setDate] = useState(() => isoOf(new Date()))
    97→    97→  const [monthCursor, setMonthCursor] = useState(() => startOfMonth(new Date()))
    98→    98→  const [slots, setSlots] = useState<AvailabilitySlot[]>([])
    99→    99→  const [slotsLoading, setSlotsLoading] = useState(false)
   100→   100→  const [monthSummary, setMonthSummary] = useState<Record<string, number> | null>(null)
   101→   101→  const [selection, setSelection] = useState<Selection | null>(null)
   102→   102→  const [customerName, setCustomerName] = useState('')
   103→   103→  const [customerPhone, setCustomerPhone] = useState('')
   104→   104→  const [submitting, setSubmitting] = useState(false)
   105→   105→  const [created, setCreated] = useState<Reservation | null>(null)
   106→   106→  const [waveLink, setWaveLink] = useState<string | null>(null)
   107→   107→
   108→   108→  const fetchSeq = useRef(0)
   109→   109→
   110→   110→  const facility = useMemo(
   111→   111→    () => facilities.find((f) => f.id === facilityId),
   112→   112→    [facilities, facilityId],
   113→   113→  )
   114→   114→  const todayIso = isoOf(new Date())
   115→   115→  const maxIso = useMemo(() => {
   116→   116→    const d = new Date()
   117→   117→    d.setDate(d.getDate() + MAX_DAYS_AHEAD)
   118→   118→    return isoOf(d)
   119→   119→  }, [])
   120→   120→
   121→   121→  // ===== Ouverture : terrain présélectionné + réinitialisation =====
   122→   122→  useEffect(() => {
   123→   123→    if (open) {
   124→   124→      setStep('slot')
   125→   125→      setSelection(null)
   126→   126→      setCreated(null)
   127→   127→      setDate(isoOf(new Date()))
   128→   128→      setMonthCursor(startOfMonth(new Date()))
   129→   129→      setFacilityId(preselectedFacilityId ?? facilities[0]?.id ?? '')
   130→   130→      // Lien de paiement Wave (configuré par l'admin)
   131→   131→      apiFetch<PublicSettings>('/api/settings')
   132→   132→        .then((s) => setWaveLink(s.wavePaymentLink))
   133→   133→        .catch(() => setWaveLink(null))
   134→   134→    }
   135→   135→  }, [open, preselectedFacilityId, facilities])
   136→   136→
   137→   137→  // ===== Créneaux du jour sélectionné =====
   138→   138→  const loadSlots = useCallback(async (fid: string, day: string) => {
   139→   139→    if (!fid || !day) return
   140→   140→    const seq = ++fetchSeq.current
   141→   141→    setSlotsLoading(true)
   142→   142→    try {
   143→   143→      const data = await apiFetch<DayAvailability>(
   144→   144→        `/api/availability?facilityId=${encodeURIComponent(fid)}&date=${day}`,
   145→   145→      )
   146→   146→      if (seq === fetchSeq.current) setSlots(data.slots)
   147→   147→    } catch {
   148→   148→      if (seq === fetchSeq.current) setSlots([])
   149→   149→    } finally {
   150→   150→      if (seq === fetchSeq.current) setSlotsLoading(false)
   151→   151→    }
   152→   152→  }, [])
   153→   153→
   154→   154→  useEffect(() => {
   155→   155→    if (open) {
   156→   156→      setSelection(null)
   157→   157→      loadSlots(facilityId, date)
   158→   158→    }
   159→   159→  }, [open, facilityId, date, loadSlots])
   160→   160→
   161→   161→  // ===== Résumé mensuel (calendrier : jours complets / bientôt complets) =====
   162→   162→  useEffect(() => {
   163→   163→    if (!open || !facilityId) return
   164→   164→    let cancelled = false
   165→   165→    apiFetch<MonthAvailability>(
   166→   166→      `/api/availability?facilityId=${encodeURIComponent(facilityId)}&month=${monthKey(monthCursor)}`,
   167→   167→    )
   168→   168→      .then((data) => {
   169→   169→        if (!cancelled) setMonthSummary(data.days)
   170→   170→      })
   171→   171→      .catch(() => {
   172→   172→        if (!cancelled) setMonthSummary(null)
   173→   173→      })
   174→   174→    return () => {
   175→   175→      cancelled = true
   176→   176→    }
   177→   177→  }, [open, facilityId, monthCursor])
   178→   178→
   179→   179→  // ===== Sélection de créneaux consécutifs =====
   180→   180→  function toggleSlot(idx: number) {
   181→   181→    const slot = slots[idx]
   182→   182→    if (!slot || slot.state !== 'FREE') return
   183→   183→    if (!selection) {
   184→   184→      setSelection({ start: idx, end: idx })
   185→   185→      return
   186→   186→    }
   187→   187→    if (idx >= selection.start && idx <= selection.end) {
   188→   188→      if (idx === selection.start && idx === selection.end) setSelection(null)
   189→   189→      else if (idx === selection.start) setSelection({ start: idx + 1, end: selection.end })
   190→   190→      else if (idx === selection.end) setSelection({ start: selection.start, end: idx - 1 })
   191→   191→      return
   192→   192→    }
   193→   193→    if (idx === selection.start - 1) setSelection({ start: idx, end: selection.end })
   194→   194→    else if (idx === selection.end + 1) setSelection({ start: selection.start, end: idx })
   195→   195→    else setSelection({ start: idx, end: idx })
   196→   196→  }
   197→   197→
   198→   198→  const selectedSlots = useMemo(
   199→   199→    () => (selection ? slots.slice(selection.start, selection.end + 1) : []),
   200→   200→    [selection, slots],
   201→   201→  )
   202→   202→  const hours = selection ? selection.end - selection.start + 1 : 0
   203→   203→  const total = facility && hours > 0 ? hours * facility.pricePerHour : 0
   204→   204→  const canContinue = hours > 0 && !!facility
   205→   205→
   206→   206→  // ===== Jours du calendrier =====
   207→   207→  const calendarDays = useMemo(() => {
   208→   208→    const start = startOfWeek(startOfMonth(monthCursor), { weekStartsOn: 1 })
   209→   209→    const end = endOfWeek(endOfMonth(monthCursor), { weekStartsOn: 1 })
   210→   210→    return eachDayOfInterval({ start, end })
   211→   211→  }, [monthCursor])
   212→   212→
   213→   213→  const prevMonthDisabled = isSameMonth(monthCursor, new Date())
   214→   214→  const nextMonthDisabled = isSameMonth(monthCursor, addMonths(new Date(), 2))
   215→   215→
   216→   216→  // ===== Soumission =====
   217→   217→  const canSubmit =
   218→   218→    step === 'details' &&
   219→   219→    !!facility &&
   220→   220→    hours > 0 &&
   221→   221→    customerName.trim().length > 1 &&
   222→   222→    customerPhone.replace(/\D/g, '').length >= 8
   223→   223→
   224→   224→  async function handleSubmit(e: React.FormEvent) {
   225→   225→    e.preventDefault()
   226→   226→    if (!canSubmit || submitting || !selection || !facility) return
   227→   227→    const startSlot = slots[selection.start]
   228→   228→    const endSlot = slots[selection.end]
   229→   229→    if (!startSlot || !endSlot) return
   230→   230→    setSubmitting(true)
   231→   231→    try {
   232→   232→      const result = await apiFetch<{ reservation: Reservation }>('/api/reservations', {
   233→   233→        method: 'POST',
   234→   234→        body: {
   235→   235→          facilityId: facility.id,
   236→   236→          date,
   237→   237→          startTime: startSlot.start,
   238→   238→          endTime: endSlot.end,
   239→   239→          customerName: customerName.trim(),
   240→   240→          customerPhone: customerPhone.trim(),
   241→   241→          source: 'PUBLIC',
   242→   242→        },
   243→   243→      })
   244→   244→      setCreated(result.reservation)
   245→   245→      setStep('done')
   246→   246→      toast({
   247→   247→        title: 'Réservation enregistrée ! 🎉',
   248→   248→        description: `Référence ${result.reservation.reference.slice(0, 10).toUpperCase()}. Effectuez le paiement Wave pour la confirmer.`,
   249→   249→      })
   250→   250→    } catch (error) {
   251→   251→      const message = error instanceof ApiError ? error.message : 'Une erreur est survenue. Réessayez.'
   252→   252→      toast({ title: 'Échec de la réservation', description: message, variant: 'destructive' })
   253→   253→      // Conflit ou créneau devenu indisponible → retour au choix du créneau
   254→   254→      if (error instanceof ApiError && (error.status === 409 || error.status === 400)) {
   255→   255→        setStep('slot')
   256→   256→        setSelection(null)
   257→   257→        loadSlots(facilityId, date)
   258→   258→      }
   259→   259→    } finally {
   260→   260→      setSubmitting(false)
   261→   261→    }
   262→   262→  }
   263→   263→
   264→   264→  function payWithWave() {
   265→   265→    if (!waveLink || !created) return
   266→   266→    const url = buildWaveUrl(waveLink, { amount: created.amount ?? 0, reference: created.reference })
   267→   267→    window.open(url, '_blank', 'noopener')
   268→   268→  }
   269→   269→
   270→   270→  // ===== Rendu =====
   271→   271→  const stepTitles: Record<Step, { title: string; desc: string }> = {
   272→   272→    slot: {
   273→   273→      title: 'Réserver votre créneau',
   274→   274→      desc: 'Choisissez le terrain, la date puis votre créneau horaire (08:00 → minuit).',
   275→   275→    },
   276→   276→    details: {
   277→   277→      title: 'Vos coordonnées',
   278→   278→      desc: 'Laissez votre nom et votre numéro de téléphone, puis payez avec Wave.',
   279→   279→    },
   280→   280→    done: { title: 'Réservation enregistrée', desc: 'Votre créneau est bloqué, il ne reste qu’à payer.' },
   281→   281→  }
   282→   282→
   283→   283→  const freeCount = slots.filter((s) => s.state === 'FREE').length
   284→   284→  const noSlotsLeft = !slotsLoading && slots.length > 0 && freeCount === 0
   285→   285→
   286→   286→  return (
   287→   287→    <Dialog open={open} onOpenChange={onOpenChange}>
   288→   288→      <DialogContent className="sm:max-w-lg zalspor-scroll max-h-[92vh] overflow-y-auto">
   289→   289→        <DialogHeader>
   290→   290→          <DialogTitle className="flex items-center gap-2 text-xl">
   291→   291→            <CalendarCheck className="size-5 text-primary" />
   292→   292→            {stepTitles[step].title}
   293→   293→          </DialogTitle>
   294→   294→          <DialogDescription>{stepTitles[step].desc}</DialogDescription>
   295→   295→        </DialogHeader>
   296→   296→
   297→   297→        {/* ============ ÉTAPE 1 : terrain + calendrier + créneaux ============ */}
   298→   298→        {step === 'slot' && (
   299→   299→          <div className="grid gap-5">
   300→   300→            {facilities.length > 1 ? (
   301→   301→              <div className="grid grid-cols-1 gap-2">
   302→   302→                <Label>Terrain</Label>
   303→   303→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-2">
   304→   304→                  {facilities.map((f) => (
   305→   305→                    <button
   306→   306→                      key={f.id}
   307→   307→                      type="button"
   308→   308→                      onClick={() => setFacilityId(f.id)}
   309→   309→                      className={`rounded-xl border px-3 py-2.5 text-left transition-all cursor-pointer ${
   310→   310→                        facilityId === f.id
   311→   311→                          ? 'border-primary bg-primary/5 shadow-sm'
   312→   312→                          : 'hover:border-primary/40'
   313→   313→                      }`}
   314→   314→                      aria-pressed={facilityId === f.id}
   315→   315→                    >
   316→   316→                      <p className="text-sm font-semibold leading-tight truncate">{f.name.split('—')[0].trim()}</p>
   317→   317→                      <p className="text-xs text-muted-foreground mt-1">
   318→   318→                        {formatPrice(f.pricePerHour)}/h · {f.capacity} joueurs
   319→   319→                      </p>
   320→   320→                    </button>
   321→   321→                  ))}
   322→   322→                </div>
   323→   323→              </div>
   324→   324→            ) : facilities.length === 1 ? (
   325→   325→              <Card>
   326→   326→                <CardContent className="flex items-center justify-between py-3 px-4">
   327→   327→                  <span className="text-sm font-semibold">{facilities[0].name}</span>
   328→   328→                  <span className="text-sm text-primary font-bold">{formatPrice(facilities[0].pricePerHour)}/h</span>
   329→   329→                </CardContent>
   330→   330→              </Card>
   331→   331→            ) : (
   332→   332→              <p className="text-sm text-muted-foreground">Aucun terrain disponible pour le moment.</p>
   333→   333→            )}
   334→   334→
   335→   335→            {/* ----- Calendrier mensuel ----- */}
   336→   336→            <div className="rounded-xl border bg-card">
   337→   337→              <div className="flex items-center justify-between px-3 py-2.5 border-b">
   338→   338→                <Button
   339→   339→                  type="button"
   340→   340→                  variant="ghost"
   341→   341→                  size="icon"
   342→   342→                  className="size-8"
   343→   343→                  disabled={prevMonthDisabled}
   344→   344→                  onClick={() => setMonthCursor((m) => addMonths(m, -1))}
   345→   345→                  aria-label="Mois précédent"
   346→   346→                >
   347→   347→                  <ChevronLeft className="size-4" />
   348→   348→                </Button>
   349→   349→                <p className="text-sm font-semibold capitalize">
   350→   350→                  {format(monthCursor, 'MMMM yyyy', { locale: fr })}
   351→   351→                </p>
   352→   352→                <Button
   353→   353→                  type="button"
   354→   354→                  variant="ghost"
   355→   355→                  size="icon"
   356→   356→                  className="size-8"
   357→   357→                  disabled={nextMonthDisabled}
   358→   358→                  onClick={() => setMonthCursor((m) => addMonths(m, 1))}
   359→   359→                  aria-label="Mois suivant"
   360→   360→                >
   361→   361→                  <ChevronRight className="size-4" />
   362→   362→                </Button>
   363→   363→              </div>
   364→   364→              <div className="p-3">
   365→   365→                <div className="grid grid-cols-7 gap-1 mb-1">
   366→   366→                  {WEEKDAYS.map((d) => (
   367→   367→                    <p key={d} className="text-center text-[11px] font-medium text-muted-foreground py-1">
   368→   368→                      {d}
   369→   369→                    </p>
   370→   370→                  ))}
   371→   371→                </div>
   372→   372→                <div className="grid grid-cols-7 gap-1">
   373→   373→                  {calendarDays.map((day) => {
   374→   374→                    const iso = isoOf(day)
   375→   375→                    const inMonth = isSameMonth(day, monthCursor)
   376→   376→                    const past = iso < todayIso
   377→   377→                    const tooFar = iso > maxIso
   378→   378→                    const full = monthSummary !== null && (monthSummary[iso] ?? -1) === 0 && !past && !tooFar
   379→   379→                    const almostFull =
   380→   380→                      !full && monthSummary !== null && (monthSummary[iso] ?? 16) > 0 && (monthSummary[iso] ?? 16) <= 4
   381→   381→                    const disabled = !inMonth || past || tooFar || full
   382→   382→                    const selected = date === iso
   383→   383→                    const today = isSameDay(day, new Date())
   384→   384→                    return (
   385→   385→                      <button
   386→   386→                        key={iso}
   387→   387→                        type="button"
   388→   388→                        disabled={disabled}
   389→   389→                        onClick={() => setDate(iso)}
   390→   390→                        className={`relative h-10 rounded-lg text-sm transition-colors ${
   391→   391→                          selected
   392→   392→                            ? 'bg-primary text-primary-foreground font-bold shadow-sm cursor-pointer'
   393→   393→                            : disabled
   394→   394→                              ? 'text-muted-foreground/40 cursor-not-allowed'
   395→   395→                              : 'hover:bg-primary/10 cursor-pointer'
   396→   396→                        } ${full ? 'line-through' : ''} ${today && !selected ? 'ring-1 ring-primary/50 font-semibold' : ''}`}
   397→   397→                        aria-label={`Choisir le ${format(day, 'd MMMM', { locale: fr })}${full ? ' (complet)' : ''}`}
   398→   398→                        aria-pressed={selected}
   399→   399→                      >
   400→   400→                        {format(day, 'd')}
   401→   401→                        {almostFull && !selected ? (
   402→   402→                          <span
   403→   403→                            className="absolute bottom-1 left-1/2 -translate-x-1/2 size-1 rounded-full bg-amber-500"
   404→   404→                            aria-hidden
   405→   405→                          />
   406→   406→                        ) : null}
   407→   407→                      </button>
   408→   408→                    )
   409→   409→                  })}
   410→   410→                </div>
   411→   411→              </div>
   412→   412→            </div>
   413→   413→
   414→   414→            {/* ----- Grille des créneaux 08:00 → minuit ----- */}
   415→   415→            <div className="grid gap-2">
   416→   416→              <div className="flex items-center justify-between gap-2 min-w-0">
   417→   417→                <Label className="flex items-center gap-1.5 min-w-0">
   418→   418→                  <Clock className="size-3.5 text-primary shrink-0" />
   419→   419→                  <span className="truncate">Créneaux du {formatDateFr(date)}</span>
   420→   420→                </Label>
   421→   421→                <span className="text-xs text-muted-foreground shrink-0 whitespace-nowrap">
   422→   422→                  {slotsLoading ? '…' : `${freeCount} libre${freeCount > 1 ? 's' : ''} / ${slots.length}`}
   423→   423→                </span>
   424→   424→              </div>
   425→   425→
   426→   426→              {slotsLoading ? (
   427→   427→                <div className="grid grid-cols-4 gap-2">
   428→   428→                  {Array.from({ length: 16 }).map((_, i) => (
   429→   429→                    <Skeleton key={i} className="h-16 rounded-xl" />
   430→   430→                  ))}
   431→   431→                </div>
   432→   432→              ) : slots.length === 0 ? (
   433→   433→                <div className="rounded-xl border border-dashed py-8 text-center text-sm text-muted-foreground">
   434→   434→                  Disponibilités indisponibles pour cette date.
   435→   435→                </div>
   436→   436→              ) : (
   437→   437→                <>
   438→   438→                  <div className="grid grid-cols-4 gap-2">
   439→   439→                    {slots.map((slot, idx) => {
   440→   440→                      const isSelected =
   441→   441→                        selection !== null && idx >= selection.start && idx <= selection.end
   442→   442→                      const stateLabel =
   443→   443→                        slot.state === 'BOOKED' ? 'Réservé' : slot.state === 'PAST' ? 'Passé' : slot.state === 'CLOSED' ? 'Fermé' : null
   444→   444→                      return (
   445→   445→                        <button
   446→   446→                          key={slot.start}
   447→   447→                          type="button"
