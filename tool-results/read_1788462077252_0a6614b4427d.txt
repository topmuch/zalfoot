     1→'use client'
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
     4→import Image from 'next/image'
     5→import {
     6→  ArrowLeft,
     7→  ArrowRight,
     8→  CalendarCheck,
     9→  Check,
    10→  CheckCircle2,
    11→  ChevronLeft,
    12→  ChevronRight,
    13→  Clock,
    14→  Info,
    15→  Loader2,
    16→  Phone,
    17→  Trophy,
    18→  User,
    19→  Users,
    20→} from 'lucide-react'
    21→import {
    22→  addMonths,
    23→  eachDayOfInterval,
    24→  endOfMonth,
    25→  endOfWeek,
    26→  format,
    27→  isSameDay,
    28→  isSameMonth,
    29→  startOfMonth,
    30→  startOfWeek,
    31→} from 'date-fns'
    32→import { fr } from 'date-fns/locale'
    33→import { Button } from '@/components/ui/button'
    34→import { Badge } from '@/components/ui/badge'
    35→import { Card, CardContent } from '@/components/ui/card'
    36→import { Input } from '@/components/ui/input'
    37→import { Label } from '@/components/ui/label'
    38→import { Skeleton } from '@/components/ui/skeleton'
    39→import { useToast } from '@/hooks/use-toast'
    40→import { apiFetch, ApiError } from './api'
    41→import {
    42→  buildWaveUrl,
    43→  formatDateFr,
    44→  formatHourLabel,
    45→  formatPrice,
    46→  type AvailabilitySlot,
    47→  type DayAvailability,
    48→  type Facility,
    49→  type MonthAvailability,
    50→  type PublicSettings,
    51→  type Reservation,
    52→} from './types'
    53→import { DEPOSIT_PER_HOUR } from '../../lib/pricing'
    54→
    55→// ============================================================
    56→// Page de réservation — PAGE DÉDIÉE (plus de modale).
    57→// Calendrier visible + créneaux horaires 08:00 → minuit (1 h),
    58→// sélection d'heures consécutives, coordonnées (nom + téléphone),
    59→// puis paiement de l'ACOMPTE (5 000 FCFA / heure réservée) via
    60→// Wave Business. Le solde est réglé sur place.
    61→// 25 000 FCFA / heure.
    62→// ============================================================
    63→
    64→const WEEKDAYS = ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim']
    65→/** Réservation possible jusqu'à 60 jours à l'avance */
    66→const MAX_DAYS_AHEAD = 60
    67→
    68→function isoOf(d: Date): string {
    69→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    70→}
    71→
    72→function monthKey(d: Date): string {
    73→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
    74→}
    75→
    76→type Selection = { start: number; end: number }
    77→
    78→type Step = 'slot' | 'details' | 'done'
    79→
    80→const STEPS: { key: Step; label: string }[] = [
    81→  { key: 'slot', label: 'Créneau' },
    82→  { key: 'details', label: 'Coordonnées' },
    83→  { key: 'done', label: 'Paiement' },
    84→]
    85→
    86→export function ReservationPage({
    87→  facilities,
    88→  preselectedFacilityId,
    89→  onBack,
    90→}: {
    91→  facilities: Facility[]
    92→  preselectedFacilityId?: string
    93→  onBack: () => void
    94→}) {
    95→  const { toast } = useToast()
    96→
    97→  const [step, setStep] = useState<Step>('slot')
    98→  const [facilityId, setFacilityId] = useState(() => preselectedFacilityId ?? facilities[0]?.id ?? '')
    99→  const [date, setDate] = useState(() => isoOf(new Date()))
   100→  const [monthCursor, setMonthCursor] = useState(() => startOfMonth(new Date()))
   101→  const [slots, setSlots] = useState<AvailabilitySlot[]>([])
   102→  const [slotsLoading, setSlotsLoading] = useState(false)
   103→  const [monthSummary, setMonthSummary] = useState<Record<string, number> | null>(null)
   104→  const [selection, setSelection] = useState<Selection | null>(null)
   105→  const [customerName, setCustomerName] = useState('')
   106→  const [customerPhone, setCustomerPhone] = useState('')
   107→  const [submitting, setSubmitting] = useState(false)
   108→  const [created, setCreated] = useState<Reservation | null>(null)
   109→  const [waveLink, setWaveLink] = useState<string | null>(null)
   110→
   111→  const fetchSeq = useRef(0)
   112→
   113→  const facility = useMemo(
   114→    () => facilities.find((f) => f.id === facilityId),
   115→    [facilities, facilityId],
   116→  )
   117→  const todayIso = isoOf(new Date())
   118→  const maxIso = useMemo(() => {
   119→    const d = new Date()
   120→    d.setDate(d.getDate() + MAX_DAYS_AHEAD)
   121→    return isoOf(d)
   122→  }, [])
   123→
   124→  // ===== Lien de paiement Wave (configuré par l'admin) =====
   125→  useEffect(() => {
   126→    apiFetch<PublicSettings>('/api/settings')
   127→      .then((s) => setWaveLink(s.wavePaymentLink))
   128→      .catch(() => setWaveLink(null))
   129→  }, [])
   130→
   131→  // ===== Créneaux du jour sélectionné =====
   132→  const loadSlots = useCallback(async (fid: string, day: string) => {
   133→    if (!fid || !day) return
   134→    const seq = ++fetchSeq.current
   135→    setSlotsLoading(true)
   136→    try {
   137→      const data = await apiFetch<DayAvailability>(
   138→        `/api/availability?facilityId=${encodeURIComponent(fid)}&date=${day}`,
   139→      )
   140→      if (seq === fetchSeq.current) setSlots(data.slots)
   141→    } catch {
   142→      if (seq === fetchSeq.current) setSlots([])
   143→    } finally {
   144→      if (seq === fetchSeq.current) setSlotsLoading(false)
   145→    }
   146→  }, [])
   147→
   148→  useEffect(() => {
   149→    setSelection(null)
   150→    loadSlots(facilityId, date)
   151→  }, [facilityId, date, loadSlots])
   152→
   153→  // ===== Résumé mensuel (calendrier : jours complets / bientôt complets) =====
   154→  useEffect(() => {
   155→    if (!facilityId) return
   156→    let cancelled = false
   157→    apiFetch<MonthAvailability>(
   158→      `/api/availability?facilityId=${encodeURIComponent(facilityId)}&month=${monthKey(monthCursor)}`,
   159→    )
   160→      .then((data) => {
   161→        if (!cancelled) setMonthSummary(data.days)
   162→      })
   163→      .catch(() => {
   164→        if (!cancelled) setMonthSummary(null)
   165→      })
   166→    return () => {
   167→      cancelled = true
   168→    }
   169→  }, [facilityId, monthCursor])
   170→
   171→  // ===== Remonter en haut de la page à chaque étape =====
   172→  useEffect(() => {
   173→    window.scrollTo({ top: 0, behavior: 'smooth' })
   174→  }, [step])
   175→
   176→  // ===== Sélection de créneaux consécutifs =====
   177→  function toggleSlot(idx: number) {
   178→    const slot = slots[idx]
   179→    if (!slot || slot.state !== 'FREE') return
   180→    if (!selection) {
   181→      setSelection({ start: idx, end: idx })
   182→      return
   183→    }
   184→    if (idx >= selection.start && idx <= selection.end) {
   185→      if (idx === selection.start && idx === selection.end) setSelection(null)
   186→      else if (idx === selection.start) setSelection({ start: idx + 1, end: selection.end })
   187→      else if (idx === selection.end) setSelection({ start: selection.start, end: idx - 1 })
   188→      return
   189→    }
   190→    if (idx === selection.start - 1) setSelection({ start: idx, end: selection.end })
   191→    else if (idx === selection.end + 1) setSelection({ start: selection.start, end: idx })
   192→    else setSelection({ start: idx, end: idx })
   193→  }
   194→
   195→  const selectedSlots = useMemo(
   196→    () => (selection ? slots.slice(selection.start, selection.end + 1) : []),
   197→    [selection, slots],
   198→  )
   199→  const hours = selection ? selection.end - selection.start + 1 : 0
   200→  const total = facility && hours > 0 ? hours * facility.pricePerHour : 0
   201→  /** Acompte : 5 000 FCFA par heure réservée */
   202→  const deposit = hours > 0 ? hours * DEPOSIT_PER_HOUR : 0
   203→  /** Solde à régler sur place */
   204→  const balance = Math.max(total - deposit, 0)
   205→  const canContinue = hours > 0 && !!facility
   206→
   207→  // ===== Jours du calendrier =====
   208→  const calendarDays = useMemo(() => {
   209→    const start = startOfWeek(startOfMonth(monthCursor), { weekStartsOn: 1 })
   210→    const end = endOfWeek(endOfMonth(monthCursor), { weekStartsOn: 1 })
   211→    return eachDayOfInterval({ start, end })
   212→  }, [monthCursor])
   213→
   214→  const prevMonthDisabled = isSameMonth(monthCursor, new Date())
   215→  const nextMonthDisabled = isSameMonth(monthCursor, addMonths(new Date(), 2))
   216→
   217→  // ===== Soumission =====
   218→  const canSubmit =
   219→    step === 'details' &&
   220→    !!facility &&
   221→    hours > 0 &&
   222→    customerName.trim().length > 1 &&
   223→    customerPhone.replace(/\D/g, '').length >= 8
   224→
   225→  async function handleSubmit(e: React.FormEvent) {
   226→    e.preventDefault()
   227→    if (!canSubmit || submitting || !selection || !facility) return
   228→    const startSlot = slots[selection.start]
   229→    const endSlot = slots[selection.end]
   230→    if (!startSlot || !endSlot) return
   231→    setSubmitting(true)
   232→    try {
   233→      const result = await apiFetch<{ reservation: Reservation }>('/api/reservations', {
   234→        method: 'POST',
   235→        body: {
   236→          facilityId: facility.id,
   237→          date,
   238→          startTime: startSlot.start,
   239→          endTime: endSlot.end,
   240→          customerName: customerName.trim(),
   241→          customerPhone: customerPhone.trim(),
   242→          source: 'PUBLIC',
   243→        },
   244→      })
   245→      setCreated(result.reservation)
   246→      setStep('done')
   247→      toast({
   248→        title: 'Réservation enregistrée ! 🎉',
   249→        description: `Référence ${result.reservation.reference.slice(0, 10).toUpperCase()}. Versez l'acompte de ${formatPrice(result.reservation.depositAmount ?? deposit)} via Wave pour la confirmer.`,
   250→      })
   251→    } catch (error) {
   252→      const message = error instanceof ApiError ? error.message : 'Une erreur est survenue. Réessayez.'
   253→      toast({ title: 'Échec de la réservation', description: message, variant: 'destructive' })
   254→      // Conflit ou créneau devenu indisponible → retour au choix du créneau
   255→      if (error instanceof ApiError && (error.status === 409 || error.status === 400)) {
   256→        setStep('slot')
   257→        setSelection(null)
   258→        loadSlots(facilityId, date)
   259→      }
   260→    } finally {
   261→      setSubmitting(false)
   262→    }
   263→  }
   264→
   265→  // ===== Paiement Wave : acompte de 5 000 F / heure =====
   266→  const depositToPay = created?.depositAmount ?? deposit
   267→  const totalCreated = created?.amount ?? total
   268→  const balanceToPay = Math.max(totalCreated - depositToPay, 0)
   269→
   270→  function payWithWave() {
   271→    if (!waveLink || !created) return
   272→    const url = buildWaveUrl(waveLink, { amount: depositToPay, reference: created.reference })
   273→    window.open(url, '_blank', 'noopener')
   274→  }
   275→
   276→  /** Nouvelle réservation : réinitialise le parcours. */
   277→  function reset() {
   278→    const today = isoOf(new Date())
   279→    setStep('slot')
   280→    setSelection(null)
   281→    setCreated(null)
   282→    setCustomerName('')
   283→    setCustomerPhone('')
   284→    setDate(today)
   285→    setMonthCursor(startOfMonth(new Date()))
   286→    loadSlots(facilityId, today)
   287→  }
   288→
   289→  // ===== Rendu =====
   290→  const freeCount = slots.filter((s) => s.state === 'FREE').length
   291→  const noSlotsLeft = !slotsLoading && slots.length > 0 && freeCount === 0
   292→  const stepIndex = step === 'slot' ? 0 : step === 'details' ? 1 : 2
   293→
   294→  const slotRange =
   295→    selection && slots.length > 0
   296→      ? `${slots[selection.start]?.start} → ${formatHourLabel(slots[selection.end]?.end ?? '')}`
   297→      : null
   298→
   299→  return (
   300→    <div>
   301→      {/* ===== En-tête de la page ===== */}
   302→      <section className="relative overflow-hidden border-b bg-gradient-to-br from-emerald-50 via-background to-teal-50 dark:from-emerald-950/40 dark:via-background dark:to-teal-950/30">
   303→        <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8 py-8 sm:py-12">
   304→          <Button
   305→            type="button"
   306→            variant="ghost"
   307→            size="sm"
   308→            className="text-muted-foreground hover:text-foreground -ml-2"
   309→            onClick={onBack}
   310→          >
   311→            <ArrowLeft className="size-4" />
   312→            Retour à l&apos;accueil
   313→          </Button>
   314→
   315→          <div className="mt-4 flex flex-col lg:flex-row lg:items-end justify-between gap-6">
   316→            <div>
   317→              <h1 className="text-3xl sm:text-4xl font-extrabold tracking-tight">
   318→                Réserver votre <span className="text-primary">terrain de football</span>
   319→              </h1>
   320→              <p className="text-muted-foreground mt-3 max-w-2xl leading-relaxed">
   321→                Choisissez la date et l&apos;heure (08:00 → minuit) parmi les créneaux libres, puis
   322→                bloquez votre terrain avec un acompte de{' '}
   323→                <strong className="text-foreground">{formatPrice(DEPOSIT_PER_HOUR)}</strong> par
   324→                heure réservée, payé avec Wave.
   325→              </p>
   326→              <div className="flex flex-wrap gap-2 mt-4">
   327→                <Badge variant="outline" className="gap-1.5">
   328→                  <Clock className="size-3.5" /> 25 000 FCFA / heure
   329→                </Badge>
   330→                <Badge variant="outline" className="gap-1.5 bg-[#00A0E7]/5 border-[#00A0E7]/30 text-[#0090D2] dark:text-[#4DC3F0]">
   331→                  <span className="flex size-3.5 items-center justify-center rounded-[4px] bg-white ring-1 ring-black/10 overflow-hidden">
   332→                    <Image src="/wave-logo.png" alt="" width={14} height={14} />
   333→                  </span>
   334→                  Acompte {formatPrice(DEPOSIT_PER_HOUR)}/h via Wave
   335→                </Badge>
   336→                <Badge variant="outline" className="gap-1.5">
   337→                  <Users className="size-3.5" /> Solde sur place
   338→                </Badge>
   339→              </div>
   340→            </div>
   341→
   342→            {/* Stepper */}
   343→            <ol className="flex items-center gap-2 sm:gap-3 w-full lg:w-auto lg:min-w-[380px]" aria-label="Étapes de réservation">
   344→              {STEPS.map((s, i) => {
   345→                const done = i < stepIndex
   346→                const active = i === stepIndex
   347→                const clickable = done && step !== 'slot'
   348→                return (
   349→                  <li key={s.key} className="flex items-center gap-2 sm:gap-3 flex-1 lg:flex-none min-w-0">
   350→                    <button
   351→                      type="button"
   352→                      disabled={!clickable}
   353→                      onClick={() => setStep(s.key)}
   354→                      className={`flex items-center gap-2 min-w-0 rounded-lg px-1 py-1 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary ${
   355→                        clickable ? 'cursor-pointer' : 'cursor-default'
   356→                      }`}
   357→                      aria-current={active ? 'step' : undefined}
   358→                    >
   359→                      <span
   360→                        className={`flex size-7 sm:size-8 shrink-0 items-center justify-center rounded-full text-xs font-bold border-2 transition-colors ${
   361→                          active
   362→                            ? 'bg-primary border-primary text-primary-foreground shadow-sm'
   363→                            : done
   364→                              ? 'bg-primary/10 border-primary text-primary'
   365→                              : 'bg-muted/60 border-border text-muted-foreground/70'
   366→                        }`}
   367→                      >
   368→                        {done ? <Check className="size-4" /> : i + 1}
   369→                      </span>
   370→                      <span
   371→                        className={`text-xs sm:text-sm font-semibold truncate ${
   372→                          active ? 'text-foreground' : 'text-muted-foreground'
   373→                        }`}
   374→                      >
   375→                        {s.label}
   376→                      </span>
   377→                    </button>
   378→                    {i < STEPS.length - 1 && (
   379→                      <span
   380→                        className={`h-0.5 flex-1 lg:w-8 rounded ${done ? 'bg-primary/40' : 'bg-border'}`}
   381→                        aria-hidden
   382→                      />
   383→                    )}
   384→                  </li>
   385→                )
   386→              })}
   387→            </ol>
   388→          </div>
   389→        </div>
   390→      </section>
   391→
   392→      {/* ===== Contenu ===== */}
   393→      <section className="py-8 sm:py-12">
   394→        <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8">
   395→          {facilities.length === 0 ? (
   396→            <Card>
   397→              <CardContent className="py-16 text-center text-muted-foreground">
   398→                Aucun terrain n&apos;est disponible pour le moment. Réessayez bientôt.
   399→              </CardContent>
   400→            </Card>
   401→          ) : step === 'slot' ? (
   402→            /* ============ ÉTAPE 1 : terrain + calendrier + créneaux ============ */
   403→            <div className="grid gap-6 lg:grid-cols-[minmax(0,5fr)_minmax(0,7fr)]">
   404→              {/* ----- Colonne gauche : terrain + calendrier ----- */}
   405→              <div className="grid gap-6 content-start">
   406→                <div className="grid gap-2">
   407→                  <Label className="flex items-center gap-1.5">
   408→                    <Trophy className="size-3.5 text-primary" />
   409→                    Terrain
   410→                  </Label>
   411→                  <div className="grid gap-2">
   412→                    {facilities.map((f) => (
   413→                      <button
   414→                        key={f.id}
   415→                        type="button"
   416→                        onClick={() => setFacilityId(f.id)}
   417→                        className={`rounded-xl border px-4 py-3 text-left transition-all cursor-pointer ${
   418→                          facilityId === f.id
   419→                            ? 'border-primary bg-primary/5 shadow-sm'
   420→                            : 'hover:border-primary/40'
   421→                        }`}
   422→                        aria-pressed={facilityId === f.id}
   423→                      >
   424→                        <p className="text-sm font-semibold leading-tight">{f.name}</p>
   425→                        <p className="text-xs text-muted-foreground mt-1 flex items-center gap-3">
   426→                          <span className="flex items-center gap-1">
   427→                            <Clock className="size-3 text-primary" />
   428→                            {formatPrice(f.pricePerHour)}/h
   429→                          </span>
   430→                          <span className="flex items-center gap-1">
   431→                            <Users className="size-3" />
   432→                            {f.capacity} joueurs
   433→                          </span>
   434→                        </p>
   435→                      </button>
   436→                    ))}
   437→                  </div>
   438→                </div>
   439→
   440→                {/* Calendrier mensuel */}
   441→                <div className="rounded-xl border bg-card">
   442→                  <div className="flex items-center justify-between px-3 py-2.5 border-b">
   443→                    <Button
   444→                      type="button"
   445→                      variant="ghost"
   446→                      size="icon"
   447→                      className="size-8"
   448→                      disabled={prevMonthDisabled}
   449→                      onClick={() => setMonthCursor((m) => addMonths(m, -1))}
   450→                      aria-label="Mois précédent"
   451→                    >
   452→                      <ChevronLeft className="size-4" />
   453→                    </Button>
   454→                    <p className="text-sm font-semibold capitalize">
   455→                      {format(monthCursor, 'MMMM yyyy', { locale: fr })}
   456→                    </p>
   457→                    <Button
   458→                      type="button"
   459→                      variant="ghost"
   460→                      size="icon"
   461→                      className="size-8"
   462→                      disabled={nextMonthDisabled}
   463→                      onClick={() => setMonthCursor((m) => addMonths(m, 1))}
   464→                      aria-label="Mois suivant"
   465→                    >
   466→                      <ChevronRight className="size-4" />
   467→                    </Button>
   468→                  </div>
   469→                  <div className="p-3">
   470→                    <div className="grid grid-cols-7 gap-1 mb-1">
   471→                      {WEEKDAYS.map((d) => (
   472→                        <p key={d} className="text-center text-[11px] font-medium text-muted-foreground py-1">
   473→                          {d}
   474→                        </p>
   475→                      ))}
   476→                    </div>
   477→                    <div className="grid grid-cols-7 gap-1">
   478→                      {calendarDays.map((day) => {
   479→                        const iso = isoOf(day)
   480→                        const inMonth = isSameMonth(day, monthCursor)
   481→                        const past = iso < todayIso
   482→                        const tooFar = iso > maxIso
   483→                        const full = monthSummary !== null && (monthSummary[iso] ?? -1) === 0 && !past && !tooFar
   484→                        const almostFull =
   485→                          !full && monthSummary !== null && (monthSummary[iso] ?? 16) > 0 && (monthSummary[iso] ?? 16) <= 4
   486→                        const disabled = !inMonth || past || tooFar || full
   487→                        const selected = date === iso
   488→                        const today = isSameDay(day, new Date())
   489→                        return (
   490→                          <button
   491→                            key={iso}
   492→                            type="button"
   493→                            disabled={disabled}
   494→                            onClick={() => setDate(iso)}
   495→                            className={`relative h-10 rounded-lg text-sm transition-colors ${
   496→                              selected
   497→                                ? 'bg-primary text-primary-foreground font-bold shadow-sm cursor-pointer'
   498→                                : disabled
   499→                                  ? 'text-muted-foreground/40 cursor-not-allowed'
   500→                                  : 'hover:bg-primary/10 cursor-pointer'
   501→                            } ${full ? 'line-through' : ''} ${today && !selected ? 'ring-1 ring-primary/50 font-semibold' : ''}`}
   502→                            aria-label={`Choisir le ${format(day, 'd MMMM', { locale: fr })}${full ? ' (complet)' : ''}`}
   503→                            aria-pressed={selected}
   504→                          >
   505→                            {format(day, 'd')}
   506→                            {almostFull && !selected ? (
   507→                              <span
   508→                                className="absolute bottom-1 left-1/2 -translate-x-1/2 size-1 rounded-full bg-amber-500"
   509→                                aria-hidden
   510→                              />
   511→                            ) : null}
   512→                          </button>
   513→                        )
   514→                      })}
   515→                    </div>
   516→                  </div>
   517→                </div>
   518→              </div>
   519→
   520→              {/* ----- Colonne droite : créneaux + récapitulatif ----- */}
   521→              <div className="grid gap-6 content-start">
   522→                <Card>
   523→                  <CardContent className="p-5 grid gap-4">
